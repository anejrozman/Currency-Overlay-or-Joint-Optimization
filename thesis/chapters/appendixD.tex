\chapter{Coherent Risk Measures and Deviation Measures} \label{app:demeaning}

In this appendix, we define the notion of a deviation measure and a coherent risk measure, and prove that a strictly expectation bounded coherent risk measure can be transformed into a deviation measure by a simple de-meaning operation. This result is used in \Cref{sec:mechanism-level} to de-mean expected shortfall and second-order expected shortfall risk measures and transform them into deviation measures. Throughout this appendix, let $X \in L^2(\Omega)$ denote a square-integrable
random variable on a probability space $(\Omega, \mathcal{F}, \mathbb{P})$.

\begin{definition}[Deviation measure]\label{def:deviation}
A functional $\mathcal{D} : L^2(\Omega) \to [0, \infty]$ is a \emph{deviation measure} if for all
$X, X' \in L^2(\Omega)$, every constant $C$ and every $\lambda > 0$,
\begin{align*}
&\text{(D1)} && \mathcal{D}(X + C) = \mathcal{D}(X), \\
&\text{(D2)} && \mathcal{D}(0) = 0 \quad \text{and} \quad \mathcal{D}(\lambda X) = \lambda \mathcal{D}(X), \\
&\text{(D3)} && \mathcal{D}(X + X') \leq \mathcal{D}(X) + \mathcal{D}(X'), \\
&\text{(D4)} && \mathcal{D}(X) \geq 0, \text{ with } \mathcal{D}(X) > 0 \text{ for nonconstant } X .
\end{align*}
\end{definition}


The property \emph{(D1)} is the main one that distinguishes deviation measures from coherent risk measures. It is a form of translation invariance. 

\begin{definition}[Coherent and strictly expectation bounded risk measure]\label{def:coherent}
A functional $\mathcal{R} : L^2(\Omega) \to (-\infty, \infty]$ is a \emph{coherent risk measure} if
\begin{align*}
&\text{(R1)} && \mathcal{R}(X + C) = \mathcal{R}(X) - C, \\
&\text{(R2)} && \mathcal{R}(0) = 0 \quad \text{and} \quad \mathcal{R}(\lambda X) = \lambda \mathcal{R}(X) \text{ for } \lambda > 0, \\
&\text{(R3)} && \mathcal{R}(X + X') \leq \mathcal{R}(X) + \mathcal{R}(X'), \\
&\text{(R4)} && \mathcal{R}(X) \leq \mathcal{R}(X') \text{ whenever } X \geq X' ,
\end{align*}
and it is \emph{strictly expectation bounded} if in addition
\begin{align*}
&\text{(R5)} && \mathcal{R}(X) > \mathbb{E}[-X] \quad \text{for every nonconstant }X,\quad and \quad \mathcal{R}(X) = \mathbb{E}[-X] \text{ for constant } X.
\end{align*}
\end{definition}

As mentioned above, properties \emph{(R1)} and \emph{(D1)} are incompatible, so no non-degenerate functional can be a risk measure and a deviation
measure at once. The de-meaning transformation is what moves between the two classes. We state and prove only the operation for transforming a coherent and strictly expectation bounded  risk measure into a deviation measure. This is one direction of the more general result of Theorem 2 in \cite{RockafellarUryasevZabarankin2006}.

\begin{theorem}\label{thm:demeaning}
Let $\mathcal{R}$ be a strictly expectation bounded coherent risk measure. Then
\begin{equation}
\mathcal{D}(X) \coloneqq \mathcal{R}\big(X - \mathbb{E}[X]\big)
\label{eq:demeaning}
\end{equation}
is a deviation measure.
\end{theorem}

\begin{proof}
The proof is a straightforward verification of the four properties in Definition \ref{def:deviation}.
Write $\bar X = X - \mathbb{E}[X]$, then $\mathcal{D}(X) = \mathcal{R}(\bar X)$.

\emph{(D1):} For any constant $C$ we have $(X + C) - \mathbb{E}[X + C] = \bar X$, hence
$\mathcal{D}(X + C) = \mathcal{R}(\bar X) = \mathcal{D}(X)$.

\emph{(D2):} Taking $X = 0$ gives $\mathcal{D}(0) = \mathcal{R}(0) = 0$ by \emph{(R2)}. For $\lambda > 0$,
$\lambda X - \mathbb{E}[\lambda X] = \lambda \bar X$, so again, \emph{(R2)} gives
$\mathcal{D}(\lambda X) = \mathcal{R}(\lambda \bar X) = \lambda \mathcal{R}(\bar X) = \lambda \mathcal{D}(X)$.

\emph{(D3):} Since $(X + X') - \mathbb{E}[X + X'] = \bar X + \bar X'$, \emph{(R3)} gives
$\mathcal{D}(X + X') = \mathcal{R}(\bar X + \bar X') \leq \mathcal{R}(\bar X) + \mathcal{R}(\bar X')
= \mathcal{D}(X) + \mathcal{D}(X')$.

\emph{(D4):} If $X$ is constant then $\bar X = 0$ and $\mathcal{D}(X) = 0$. If $X$ is nonconstant then so
is $\bar X$, and \emph{(R5)} applied to $\bar X$ gives
$\mathcal{D}(X) = \mathcal{R}(\bar X) > \mathbb{E}[-\bar X] = 0$.
\end{proof}

The proof uses the properties \emph{(R2)}, \emph{(R3)} and \emph{(R5)} only. Monotonicity \emph{(R4)} is never used, which is the reason the
original Theorem in \cite{RockafellarUryasevZabarankin2006} is stated more genetally for strictly expectation bounded risk
measures rather than coherent risk measures.

We now show that both expected shortfall and second-order expected shortfall are strictly expectation bounded coherent risk measures, and we can therefore apply    
\Cref{thm:demeaning} to transform them into deviation measures. Write $z^{+} = \max\{z, 0\}$ and $z^{-} = \max\{-z, 0\}$, fix a confidence level
$\alpha \in (0,1)$ and set $\kappa \coloneqq (1-\alpha)^{-1} > 1$. 
In the Rockafellar--Uryasev form \cite{RockafellarUryasev2002} the expected shortfall is
\begin{equation}
\mathrm{ES}_\alpha(X) = \min_{\eta \in \mathbb{R}} \varphi(X, \eta),
\label{eq:es-ru-form} 
\end{equation}
and its second-order analogue, obtained by replacing the expectation in $\varphi$ with the
$L^2$ norm \cite{Krokhmal2007}, is
\begin{equation}
\mathrm{SOES}_\alpha(X) = \min_{\eta \in \mathbb{R}} \psi(X, \eta),
\label{eq:soes-ru-form}
\end{equation}
where
\begin{equation*}
\varphi(X, \eta) = \eta + \kappa\, \mathbb{E}\big[(-X-\eta)^{+}\big],
\qquad
\psi(X, \eta) = \eta + \kappa\, \big\| (-X-\eta)^{+} \big\|_2 .
\end{equation*}
Both $\varphi(X, \cdot)$ and $\psi(X, \cdot)$ are convex, and both diverge as $|\eta| \to \infty$, so
the minima in \eqref{eq:es-ru-form} and \eqref{eq:soes-ru-form} are attained. Appendix \ref{app:es_auxiliary_formulation} provides additional details on the Rockafellar--Uryasev formulation. It is well known that $\mathrm{ES}_\alpha$ is a coherent risk measure, see, for example Proposition 3.1 and Corollary 4.3 in \cite{acerbi2002coherenceexpectedshortfall}. We focus on showing it is strictly expectation bounded. 

\begin{proposition}[Expected Shortfall]\label{prop:es-seb}
$\mathrm{ES}_\alpha$ is strictly expectation bounded.
\end{proposition}

\begin{proof}
Fix $\eta$, write $\mu = \mathbb{E}[X]$ and put $Y = -X - \eta$, so that
$\mathbb{E}[Y] = -\mu - \eta$ and hence $\eta + \mu = -\mathbb{E}[Y]$. Using
$\mathbb{E}[Y] = \mathbb{E}[Y^{+}] - \mathbb{E}[Y^{-}]$ we write
\begin{align*}
\varphi(X, \eta) - \mathbb{E}[-X]
&= \eta + \mu + \kappa\, \mathbb{E}[Y^{+}] \\
&= -\mathbb{E}[Y] + \kappa\, \mathbb{E}[Y^{+}]\\
&= \mathbb{E}[Y^{-}] + (\kappa - 1)\,\mathbb{E}[Y^{+}] .
\end{align*}
Both terms on the right are non-negative, the second because $\kappa > 1$, so
$\varphi(X, \eta) \geq \mathbb{E}[-X]$ for every $\eta$, with equality if and only if
$Y = 0$ almost surely, that is $X \equiv -\eta$. If $X$ is nonconstant the inequality is
therefore strict at every $\eta$, and since the
minimum is attained, $\mathrm{ES}_\alpha(X) > \mathbb{E}[-X]$. If $X \equiv C$, then $\eta = -C$ attains equality,
so $\mathrm{ES}_\alpha(X) = \mathbb{E}[-X]$.
\end{proof}

\begin{proposition}[Second-order Expected Shortfall]\label{prop:soes-coherent}
$\mathrm{SOES}_\alpha$ is a coherent, strictly expectation bounded risk measure.
\end{proposition}

\begin{proof}
\emph{(R1):} Substituting $\eta' = \eta + C$ gives $-X - C - \eta = -X - \eta'$ and
$\psi(X + C, \eta) = \eta' - C + \kappa \|(-X-\eta')^{+}\|_2 = \psi(X, \eta') - C$. As $\eta$ ranges over
$\mathbb{R}$ so does $\eta'$, so minimizing both sides gives
$\mathrm{SOES}_\alpha(X + C) = \mathrm{SOES}_\alpha(X) - C$.

\emph{(R2):} For $X = 0$ we have $\psi(0, \eta) = \eta + \kappa\, \eta^{-}$, which equals $\eta \geq 0$ for
$\eta \geq 0$ and $(1-\kappa)\eta > 0$ for $\eta < 0$. The minimum is $0$, attained at $\eta = 0$. For
$\lambda > 0$, substituting $\eta = \lambda \eta'$ and using positive homogeneity of $z \mapsto z^{+}$
and of the norm,
\begin{equation*}
\psi(\lambda X, \lambda \eta')
= \lambda \eta' + \kappa \big\| \lambda (-X - \eta')^{+} \big\|_2
= \lambda\left[\eta' + \kappa \big\| (-X-\eta')^{+} \big\|_2\right]
= \lambda\, \psi(X, \eta') ,
\end{equation*}
and minimizing over $\eta'$ gives $\mathrm{SOES}_\alpha(\lambda X) = \lambda\, \mathrm{SOES}_\alpha(X)$.

\emph{(R3):} Let $\eta$ and $\eta'$ attain the minima for $X$ and $X'$. Since
$(a + b)^{+} \leq a^{+} + b^{+}$ pointwise, and since $\|\cdot\|_2$ is monotone on non-negative random
variables and satisfies the triangle inequality,
\begin{equation*}
\big\| \big(-(X + X') - (\eta + \eta')\big)^{+} \big\|_2
\leq \big\| (-X-\eta)^{+} + (-X'-\eta')^{+} \big\|_2
\leq \big\| (-X-\eta)^{+} \big\|_2 + \big\| (-X'-\eta')^{+} \big\|_2 .
\end{equation*}
Adding $\eta + \eta'$ and multiplying the norms by $\kappa$,
$\mathrm{SOES}_\alpha(X + X') \leq \psi(X + X', \eta + \eta') \leq \psi(X, \eta) + \psi(X', \eta')
= \mathrm{SOES}_\alpha(X) + \mathrm{SOES}_\alpha(X')$.

\emph{(R4):} If $X \geq X'$ then $-X - \eta \leq -X' - \eta$, so $(-X-\eta)^{+} \leq (-X'-\eta)^{+}$
pointwise and therefore $\psi(X, \eta) \leq \psi(X', \eta)$ for every $\eta$. Taking minima gives
$\mathrm{SOES}_\alpha(X) \leq \mathrm{SOES}_\alpha(X')$.

\emph{(R5):} On a probability space $\|Z\|_2 \geq \|Z\|_1$, so for $Z = (-X-\eta)^{+} \geq 0$ we have
$\|(-X-\eta)^{+}\|_2 \geq \mathbb{E}[(-X-\eta)^{+}]$ and hence $\psi(X, \eta) \geq \varphi(X, \eta)$ for
every $\eta$. Minimizing over $\eta$ gives $\mathrm{SOES}_\alpha(X) \geq \mathrm{ES}_\alpha(X)$, and
Proposition \ref{prop:es-seb} then gives $\mathrm{SOES}_\alpha(X) > \mathbb{E}[-X]$ for nonconstant $X$, with
equality for constant $X$ by \emph{(R1)} and \emph{(R2)}.
\end{proof}
