\chapter{Coherent Risk Measures and Deviation Measures} \label{app:demeaning}

In this appendix, we define the notions of a deviation measure and a coherent risk measure, and prove that a strictly expectation bounded coherent risk measure can be transformed into a deviation measure by a simple de-meaning operation. This result is used in \Cref{sec:mechanism-level} to de-mean the Expected Shortfall and second-order Expected Shortfall risk measures and transform them into deviation measures. Throughout this appendix, let $X \in L^2(\Omega)$ denote a square-integrable
random variable on a probability space $(\Omega, \mathcal{F}, \mathbb{P})$.

\begin{definition}[Deviation measure]\label{def:deviation}
A functional $\mathcal{D} : L^2(\Omega) \to [0, \infty]$ is a \emph{deviation measure} if for all
$X, X' \in L^2(\Omega)$, every constant $C$, and every $\lambda > 0$,
\begin{align*}
&\text{(D1)} && \mathcal{D}(X + C) = \mathcal{D}(X), \\
&\text{(D2)} && \mathcal{D}(0) = 0 \quad \text{and} \quad \mathcal{D}(\lambda X) = \lambda \mathcal{D}(X), \\
&\text{(D3)} && \mathcal{D}(X + X') \leq \mathcal{D}(X) + \mathcal{D}(X'), \\
&\text{(D4)} && \mathcal{D}(X) \geq 0, \text{ with } \mathcal{D}(X) > 0 \text{ for nonconstant } X .
\end{align*}
\end{definition}


The property \emph{(D1)} is the main one that distinguishes deviation measures from coherent risk measures. It is a form of translation invariance. 

\begin{definition}[Coherent and strictly expectation bounded risk measure]\label{def:coherent}
A functional $\mathcal{R} : L^2(\Omega) \to (-\infty, \infty]$ is a \emph{coherent risk measure} if for all
$X, X' \in L^2(\Omega)$ and every constant $C$,
\begin{align*}
&\text{(R1)} && \mathcal{R}(X + C) = \mathcal{R}(X) - C, \\
&\text{(R2)} && \mathcal{R}(0) = 0 \quad \text{and} \quad \mathcal{R}(\lambda X) = \lambda \mathcal{R}(X) \text{ for } \lambda > 0, \\
&\text{(R3)} && \mathcal{R}(X + X') \leq \mathcal{R}(X) + \mathcal{R}(X'), \\
&\text{(R4)} && \mathcal{R}(X) \leq \mathcal{R}(X') \text{ whenever } X \geq X' ,
\end{align*}
and it is \emph{strictly expectation bounded} if in addition
\begin{align*}
&\text{(R5)} && \mathcal{R}(X) > \mathbb{E}[-X] \quad \text{for every nonconstant }X,\quad \text{and} \quad \mathcal{R}(X) = \mathbb{E}[-X] \text{ for constant } X.
\end{align*}
\end{definition}

Properties \emph{(R1)} and \emph{(D1)} are incompatible, so no non-degenerate functional can be a risk measure and a deviation
measure at once. The de-meaning transformation is what moves between the two classes. We state and prove only the result for transforming a coherent and strictly expectation bounded  risk measure into a deviation measure. This is one direction of the more general result of Theorem 2 in Rockafellar et al.~\cite{RockafellarUryasevZabarankin2006}.

\begin{theorem}\label{thm:demeaning}
Let $\mathcal{R}$ be a strictly expectation bounded coherent risk measure. Then
\begin{equation}
\mathcal{D}(X) \coloneqq \mathcal{R}\big(X - \mathbb{E}[X]\big)
\end{equation}
is a deviation measure.
\end{theorem}

\begin{proof}
The proof is a straightforward verification of the four properties in Definition \ref{def:deviation}.
Write $\bar X = X - \mathbb{E}[X]$. Then $\mathcal{D}(X) = \mathcal{R}(\bar X)$.

\emph{(D1):} For any constant $C$ we have $(X + C) - \mathbb{E}[X + C] = \bar X$. Hence,
$\mathcal{D}(X + C) = \mathcal{R}(\bar X) = \mathcal{D}(X)$.

\emph{(D2):} Taking $X = 0$ gives $\mathcal{D}(0) = \mathcal{R}(0) = 0$ by \emph{(R2)}. For $\lambda > 0$,
$\lambda X - \mathbb{E}[\lambda X] = \lambda \bar X$, so again, \emph{(R2)} gives
$\mathcal{D}(\lambda X) = \mathcal{R}(\lambda \bar X) = \lambda \mathcal{R}(\bar X) = \lambda \mathcal{D}(X)$.

\emph{(D3):} Since $(X + X') - \mathbb{E}[X + X'] = \bar X + \bar X'$, \emph{(R3)} gives
$\mathcal{D}(X + X') = \mathcal{R}(\bar X + \bar X') \leq \mathcal{R}(\bar X) + \mathcal{R}(\bar X')
= \mathcal{D}(X) + \mathcal{D}(X')$.

\emph{(D4):} If $X$ is constant then $\bar X = 0$ and $\mathcal{D}(X) = 0$. If $X$ is nonconstant then so
is $\bar X$, and \emph{(R5)} applied to $\bar X$ gives
$\mathcal{D}(X) = \mathcal{R}(\bar X) > \mathbb{E}[-\bar X] = 0$.
\end{proof}

The proof uses the properties \emph{(R2)}, \emph{(R3)}, and \emph{(R5)} only. Monotonicity \emph{(R4)} is never used, which is the reason the
original theorem in Rockafellar et al.~\cite{RockafellarUryasevZabarankin2006} is stated more generally for strictly expectation bounded risk
measures rather than coherent risk measures.

We now show that both Expected Shortfall and second-order Expected Shortfall are strictly expectation bounded coherent risk measures, and we can therefore apply    
\Cref{thm:demeaning} to transform them into deviation measures. Write $z^{+} = \max\{z, 0\}$ and $z^{-} = \max\{-z, 0\}$, fix a confidence level
$\beta \in (0,1)$, and set $\kappa \coloneqq (1-\beta)^{-1} > 1$. 
In the Rockafellar--Uryasev form \cite{RockafellarUryasev2002} the Expected Shortfall is
\begin{equation}
\mathrm{ES}_\beta(X) = \min_{\alpha \in \mathbb{R}} F_\beta(X, \alpha),
\label{eq:es-ru-form} 
\end{equation}
and its second-order analogue, obtained by replacing the expectation in $F_\beta$ with the
$L^2$ norm \cite{Krokhmal2007}, is
\begin{equation}
\mathrm{SOES}_\beta(X) = \min_{\alpha \in \mathbb{R}} G_\beta(X, \alpha),
\label{eq:soes-ru-form}
\end{equation}
where
\begin{equation*}
F_\beta(X, \alpha) = \alpha + \kappa\, \mathbb{E}\big[(-X-\alpha)^{+}\big],
\qquad
G_\beta(X, \alpha) = \alpha + \kappa\, \big\| (-X-\alpha)^{+} \big\|_{L^2}
\end{equation*}
are the auxiliary functions of \Cref{app:es_auxiliary_formulation}. Both $F_\beta(X, \cdot)$ and $G_\beta(X, \cdot)$ are convex, and both diverge as $|\alpha| \to \infty$, so
the minima in \eqref{eq:es-ru-form} and \eqref{eq:soes-ru-form} are attained. It is well known that $\mathrm{ES}_\beta$ is a coherent risk measure, see, for example, Proposition 3.1 and Corollary 4.3 in Acerbi and Tasche \cite{acerbi2002coherenceexpectedshortfall}. We show it is strictly expectation bounded. 

\begin{proposition}[Expected Shortfall]\label{prop:es-seb}
$\mathrm{ES}_\beta$ is strictly expectation bounded.
\end{proposition}

\begin{proof}
Fix $\alpha$, write $\mu = \mathbb{E}[X]$, and put $Y = -X - \alpha$, so that
$\mathbb{E}[Y] = -\mu - \alpha$ and hence $\alpha + \mu = -\mathbb{E}[Y]$. Using
$\mathbb{E}[Y] = \mathbb{E}[Y^{+}] - \mathbb{E}[Y^{-}]$ we write
\begin{align*}
F_\beta(X, \alpha) - \mathbb{E}[-X]
&= \alpha + \mu + \kappa\, \mathbb{E}[Y^{+}] \\
&= -\mathbb{E}[Y] + \kappa\, \mathbb{E}[Y^{+}]\\
&= \mathbb{E}[Y^{-}] + (\kappa - 1)\,\mathbb{E}[Y^{+}] .
\end{align*}
Both terms on the right are non-negative, the second because $\kappa > 1$, so
$F_\beta(X, \alpha) \geq \mathbb{E}[-X]$ for every $\alpha$, with equality if and only if
$Y = 0$ almost surely, that is, $X \equiv -\alpha$. If $X$ is nonconstant the inequality is
therefore strict at every $\alpha$, and since the
minimum is attained, $\mathrm{ES}_\beta(X) > \mathbb{E}[-X]$. If $X \equiv C$, then $\alpha = -C$ attains equality,
so $\mathrm{ES}_\beta(X) = \mathbb{E}[-X]$.
\end{proof}

\begin{proposition}[Second-order Expected Shortfall]\label{prop:soes-coherent}
$\mathrm{SOES}_\beta$ is a coherent, strictly expectation bounded risk measure.
\end{proposition}

\begin{proof}
\emph{(R1):} Substituting $\alpha' = \alpha + C$ gives $-X - C - \alpha = -X - \alpha'$ and
$G_\beta(X + C, \alpha) = \alpha' - C + \kappa \|(-X-\alpha')^{+}\|_{L^2} = G_\beta(X, \alpha') - C$. As $\alpha$ ranges over
$\mathbb{R}$, so does $\alpha'$, so minimizing both sides gives
$\mathrm{SOES}_\beta(X + C) = \mathrm{SOES}_\beta(X) - C$.

\emph{(R2):} For $X = 0$ we have $G_\beta(0, \alpha) = \alpha + \kappa\, \alpha^{-}$, which equals $\alpha \geq 0$ for
$\alpha \geq 0$ and $(1-\kappa)\alpha > 0$ for $\alpha < 0$. The minimum is $0$, attained at $\alpha = 0$. For
$\lambda > 0$, substituting $\alpha = \lambda \alpha'$ and using positive homogeneity of $z \mapsto z^{+}$
and of the norm,
\begin{equation*}
G_\beta(\lambda X, \lambda \alpha')
= \lambda \alpha' + \kappa \big\| \lambda (-X - \alpha')^{+} \big\|_{L^2}
= \lambda\left[\alpha' + \kappa \big\| (-X-\alpha')^{+} \big\|_{L^2}\right]
= \lambda\, G_\beta(X, \alpha') ,
\end{equation*}
and minimizing over $\alpha'$ gives $\mathrm{SOES}_\beta(\lambda X) = \lambda\, \mathrm{SOES}_\beta(X)$.

\emph{(R3):} Let $\alpha$ and $\alpha'$ attain the minima for $X$ and $X'$. Since
$(a + b)^{+} \leq a^{+} + b^{+}$ pointwise, and since $\|\cdot\|_{L^2}$ is monotone on non-negative random
variables and satisfies the triangle inequality,
\begin{equation*}
\begin{aligned}
\big\| \big(-(X + X') - (\alpha + \alpha')\big)^{+} \big\|_{L^2}
&\leq \big\| (-X-\alpha)^{+} + (-X'-\alpha')^{+} \big\|_{L^2} \\
&\leq \big\| (-X-\alpha)^{+} \big\|_{L^2} + \big\| (-X'-\alpha')^{+} \big\|_{L^2} .
\end{aligned}
\end{equation*}
Adding $\alpha + \alpha'$ and multiplying the norms by $\kappa$,
$\mathrm{SOES}_\beta(X + X') \leq G_\beta(X + X', \alpha + \alpha') \leq G_\beta(X, \alpha) + G_\beta(X', \alpha')
= \mathrm{SOES}_\beta(X) + \mathrm{SOES}_\beta(X')$.

\emph{(R4):} If $X \geq X'$ then $-X - \alpha \leq -X' - \alpha$, so $(-X-\alpha)^{+} \leq (-X'-\alpha)^{+}$
pointwise and therefore $G_\beta(X, \alpha) \leq G_\beta(X', \alpha)$ for every $\alpha$. Taking the minimum gives
$\mathrm{SOES}_\beta(X) \leq \mathrm{SOES}_\beta(X')$.

\emph{(R5):} On a probability space $\|Z\|_{L^2} \geq \|Z\|_{L^1}$, so for $Z = (-X-\alpha)^{+} \geq 0$ we have
$\|(-X-\alpha)^{+}\|_{L^2} \geq \mathbb{E}[(-X-\alpha)^{+}]$ and hence $G_\beta(X, \alpha) \geq F_\beta(X, \alpha)$ for
every $\alpha$. Minimizing over $\alpha$ gives $\mathrm{SOES}_\beta(X) \geq \mathrm{ES}_\beta(X)$, and
Proposition \ref{prop:es-seb} then gives $\mathrm{SOES}_\beta(X) > \mathbb{E}[-X]$ for nonconstant $X$, with
equality for constant $X$ by \emph{(R1)} and \emph{(R2)}.
\end{proof}
